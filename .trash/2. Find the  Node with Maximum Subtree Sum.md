public class TreeNode {  
    public int data;  
    public ArrayList<TreeNode> children = new ArrayList<TreeNode>();  
  
    public TreeNode(int value){  
        data = value;  
    }  
  
    @Override  
    public String toString() {  
        return String.valueOf(this.data);  
    }  
}

import java.util.*;

	public static long SubTree(TreeNode node){  
	    long sum = 0;  
	    for(var each : node.children){  
	        sum += SubTree(each);  
	    }  
	    return (sum + node.data);  
	}  
	public static TreeNode MaximumSubTree(TreeNode node){  
	    TreeNode Maximum;  
	    long max = Integer.MIN_VALUE;  
	    int ansNode = 0;  
	  
	    for(var each : node.children){  
	        long temp = SubTree(each);  
	        if(max < temp){  
	            max = temp;  
	            ansNode = each.data;  
	        }  
	    }  
	  
	    TreeNode maximum = new TreeNode( (int)(max % (Integer.MAX_VALUE)));  
	    Maximum = new TreeNode(ansNode);  
	    Maximum.children.add(maximum);  
	  
	    return Maximum;  
	}
	  
	public static void main(String[] args) {  
	    TreeNode root = GetTree();  
  
		TreeNode node = MaximumSubTree(root);  
		Println("Maximum Sub-tree node: "+node.data+"\n Total sum :"+node.children.get(0).data);
	}
