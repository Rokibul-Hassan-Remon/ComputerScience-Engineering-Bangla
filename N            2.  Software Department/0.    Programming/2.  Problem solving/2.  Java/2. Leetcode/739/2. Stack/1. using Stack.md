class Solution {  
    public int[] finalPrices(int[] prices) {  
        Stack<Integer> stack = new Stack<>();  
        int[] result = new int[prices.length];  
  
        for(int i = 0; i < prices.length; i++){  
            while(!stack.isEmpty() && prices[stack.peek()] >= prices[i] ){  
                int previousIndx = stack.pop();  
                result[previousIndx] = prices[ previousIndx ] - prices[ i ];  
  
            }  
  
            stack.push(i);  
        }  
  
        while(!stack.isEmpty()){  
            int indx = stack.pop();  
            result[ indx ] = prices[ indx ];  
        }  
  
        return result;  
    }  
}
