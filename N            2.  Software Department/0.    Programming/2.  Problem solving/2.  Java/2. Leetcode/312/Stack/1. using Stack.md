import java.util.Stack;  
class Solution {  
    public String removeDuplicateLetters(String s) {  
        int numOfAlphabets = 26;  
       int[] frequency = new int[numOfAlphabets];  
       boolean[] isAlreadyPresent = new boolean[numOfAlphabets];  
       Stack<Character> stack = new Stack<>();  
  
       for(int i = 0; i < s.length(); i++){  
           frequency[s.charAt(i) - 'a' ]++;  
       }  
  
       for(int i = 0; i < s.length(); i++){  
            char chr = s.charAt(i);  
            int indx = chr - 'a';  
  
            frequency[indx]--;  
            if (isAlreadyPresent[indx]) continue;;  
  
            while(!stack.isEmpty() &&  stack.peek() > chr && frequency[ stack.peek() -'a' ] > 0){  
                char tempChar = stack.pop();  
                isAlreadyPresent[tempChar - 'a'] = false;  
            }  
             
            stack.push(chr);  
            isAlreadyPresent[indx] = true;  
       }  
  
       StringBuilder sb = new StringBuilder();  
        for (int i = 0; i < stack.size(); i++) {  
            sb.append(stack.get(i));  
        }  
  
        return sb.toString();  
    }  
}
