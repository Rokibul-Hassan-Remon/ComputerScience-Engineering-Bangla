public static  List<List<Integer>> combinationSum3(int k, int n) {  
    List<List<Integer>> res = new ArrayList<>();  
    List<Integer> path = new ArrayList<>();  
    int crntSum = 0;  
  
    int min = 0;  
    int max = 0;  
  
    min = k*(1+k)/2;  
    max = ((9-k)*k) + min;  
  
    if(min <= n && n <= max)  
        backtrack(k, n, 1, crntSum, path, res);  
  
   return res;  
}  
  
public static  void backtrack(int k, int n, int start, int crntSum, List<Integer> path, List<List<Integer>> res){  
    if(crntSum == n || k == path.size()){  
        if(crntSum == n && k == path.size()){  
            res.add(new ArrayList<>(path));  
        }  
  
        return;  
    }  
  
    for(int i = start; i <= 9 - ( k - path.size() ) + 1; i++){  
        if(crntSum + i > n)  
            return;  
  
        path.add(i);  
        backtrack(k , n, i+1, crntSum + i, path, res);  
        path.removeLast();  
    }  
}
