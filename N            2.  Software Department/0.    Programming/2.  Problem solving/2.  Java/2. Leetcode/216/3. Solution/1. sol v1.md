public static  List<List<Integer>> combinationSum3(int k, int n) {  
    List<List<Integer>> res = new ArrayList<>();  
    List<Integer> path = new ArrayList<>();  
    int crntSum = 0;  
  
    int min = 0;  
    int max = 0;  
  
    for(int i = 1; i <= k; i++){  
        min += i;  
    }  
  
    for(int j = 9; j >= 9-k+1; j-- ){  
        max += j;  
    }  
  
    if(min <= n && n <= max)  
        backtrack(k, n, 1, crntSum, path, res);  
  
   return res;  
}  
  
public static  void backtrack(int k, int n, int start, int crntSum, List<Integer> path, List<List<Integer>> res){  
    if(crntSum == n || k == 0){  
        if(crntSum == n && k == 0){  
            res.add(new ArrayList<>(path));  
        }  
  
        return;  
    }  
  
    for(int i = start; i <= 9 - k + 1; i++){  
        path.add(i);  
        backtrack(k - 1, n, i+1, crntSum + i, path, res);  
        path.removeLast();  
    }  
}
