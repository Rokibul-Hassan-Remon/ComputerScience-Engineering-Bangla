public List<String> restoreIpAddresses(String s) {  
    ArrayList<String> res = new ArrayList<>();  
    int n = s.length();  
  
    for(int i = 0; i < n; i++){  
        char ch = s.charAt(i);  
  
        if(!('0' <= ch && ch <= '9')){  
            return res;  
        }  
    }  
  
    int segment = 0;  
    int start = 0;  
    StringBuilder sb = new StringBuilder();  
    BackTrack(s, start, segment, sb, res);  
  
    return  res;  
}

public static void BackTrack(String s, int segment, int start, StringBuilder sb, ArrayList<String> ipList ){  
    if(segment == 4){  
        if(start == s.length()){  
            ipList.add(sb.substring(0, sb.length() -1));  
        }  
  
        return;  
    }  
  
    int remainingSegment = 4 - segment;  
    int remainingChar = s.length() - start;  
  
    if(remainingChar < remainingSegment || remainingChar > remainingSegment*3){  
        return;  
    }  
  
    for(int i = 1; i <= 3; i++){  
        if( start + i > s.length() ){  
            break;  
        }  
  
        String part = s.substring(start, start+i);  
  
        if(isValid(part) == false){  
            break;  
        }  
  
        int beforeLength = sb.length();  
        sb.append(part).append(".");  
        BackTrack(s, segment + 1, start + i, sb, ipList);  
        sb.setLength(beforeLength);  
    }  
}  
  
public static boolean isValid(String s){  
    boolean status = false;  
    int value = 0;  
  
    // s.charAt(0) == '0' or use s.startsWith("0")  
    if(s.length() > 1 && s.startsWith("0") == true){  
        return status;  
    }  
  
    value = Integer.parseInt(s);  
  
    if(value > 255){  
        return  status;  
    }  
  
    status = true;  
  
    return status;  
}
