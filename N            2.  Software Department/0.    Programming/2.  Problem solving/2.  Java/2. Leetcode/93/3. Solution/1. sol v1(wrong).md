	public static List<String> restoreIpAddresses(String s) {
        ArrayList<String> res = new ArrayList<>();
        int n = s.length();

        if(n < 4 || n > 12 ){
            return res;
        }

        for(int i = 0; i < n; i++){
            char ch = s.charAt(i);

            if(!('0' <= ch && ch <= '9')){
                return res;
            }
        }

        int segment = 4;
        Stack<Character> st = new Stack<>();

        backtrack(s, n - 1, segment, st, res);

        return  res;
    }

    public static void backtrack(String s, int indx, int segment, Stack<Character> st, List<String> ipList) {
        if (segment == 1 && indx == 0) {
            st.push(s.charAt(indx));
            StringBuilder sb = new StringBuilder();

            while (!st.isEmpty())
                sb.append(st.pop());

            if (sb.isEmpty())
                sb.append("");

            ipList.add(sb.toString());

            return;
        }

        int rstCharacter = indx + 1;
        int max = segment*3;
        int min = segment;

        if(rstCharacter >= min && rstCharacter<= max){
            if(st.isEmpty()){
                st.push(s.charAt(indx));
                st.push('.');
            }else{
               var ch = st.peek();

               if(ch == '.'){
                   st.pop();
                   st.push(s.charAt(indx));
                   if(segment != 1 )
                    st.push('.');
               }else{
                   st.push(s.charAt(indx));
               }
            }

            backtrack(s, indx-1, segment - 1, st, ipList);
        }else{
            return;
        }
    }
