public List<List<Integer>> subsets(int[] nums) {  
    List<Integer> path = new ArrayList<>();  
    List<List<Integer>> res = new ArrayList<>();  
  
    int indx = 0;  
    backtrack(nums, indx, path, res);  
  
    return res;  
}  
  
public void backtrack(int[] nums, int indx, List<Integer> path, List<List<Integer>> res){  
    if(indx == nums.length){  
        res.add(new ArrayList<>(path));  
  
        return;  
    }  
  
    backtrack(nums, indx+1, path, res);  
  
    path.add(nums[indx]);  
    backtrack(nums, indx+1, path, res);  
    path.removeLast();  
}
