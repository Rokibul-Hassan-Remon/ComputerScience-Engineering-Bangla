public static  List<List<String>> partition(String s) {  
    List<List<String>> fin = new ArrayList<>();  
    List<String> pth = new ArrayList<>();  
    backtrack(s, 0, pth, fin);  
  
    return fin;  
}  
  
public static void backtrack(String s, int indx, List<String> path, List<List<String>> result){  
    if(indx == s.length()){  
        result.add(new ArrayList<>(path));  
  
        return;  
    }  
  
    for(int i = indx; i < s.length(); i++){  
        String part = s.substring(indx, i+1);  
        if(IsPalindrom(part)){  
            path.add(part);  
            backtrack(s, i+1, path, result);  
            path.removeLast();  
        }  
    }  
}  
  
public static boolean IsPalindrom(String s){  
    int i = 0;  
    int j = s.length() - 1;  
  
    while(i < j){  
        if(s.charAt(i) != s.charAt(j))  
            return false;  
  
        i++;  
        j--;  
    }  
  
    return true;  
}
