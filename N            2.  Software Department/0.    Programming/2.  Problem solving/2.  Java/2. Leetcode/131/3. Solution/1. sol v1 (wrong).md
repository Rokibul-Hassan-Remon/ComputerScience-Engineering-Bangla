public List<List<String>> partition(String s) {  
    List<List<String>> fin = new ArrayList<>();  
    List<String> pth = new ArrayList<>();  
  
    for(int i = 0; i < s.length(); i++){  
        for(int j = i; j < s.length(); j++){  
  
            String part = s.substring(i, j+1);  
            if(IsPalindrom(part)){  
                pth.add(part);  
            }  
        }  
  
        if(pth.size() > 0){  
            fin.add(new ArrayList<>(pth));  
            pth = new ArrayList<>();  
        }  
    }  
  
    return fin;  
}  
public static boolean IsPalindrom(String s){  
    int i = 0;  
    int j = s.length() - 1;  
  
    while(i < j){  
        if(s.charAt(i) != s.charAt(j))  
            return false;  
  
        i++;  
        j--;  
    }  
  
    return true;  
}
