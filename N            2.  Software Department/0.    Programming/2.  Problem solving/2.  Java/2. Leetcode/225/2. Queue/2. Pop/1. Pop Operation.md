public class MyStack{ 

    Queue<Integer> queue;  
    
    public MyStack() {  
        queue = new ArrayDeque<>();  
    }  
  
    public void push(int x) {  
        queue.add(x);  
    }  

    public int pop() {  
        Queue<Integer> temp = new ArrayDeque<>();  
        while(queue.size() > 1){  
            int x = queue.remove();  
            temp.add(x);  
        }  
        int returnNumber = queue.remove();  
        queue = temp;  
        
        return returnNumber;  
    }  
  
    public int top() {  
        int returnNumber = pop();  
        queue.add(returnNumber);  
          
        return returnNumber;  
    }  
  
    public boolean empty() {  
        if(queue.size() > 0)  
            return false;  
             
        return true;  
    }  
}
