public class MyStack{  
    Queue<Integer> queue;  
    public MyStack() {  
        queue = new ArrayDeque<>();  
    }  
  
    public void push(int x) {  
        queue.add(x);  
  
        for(int i = 0; i < queue.size(); i++){  
            int crntVal = queue.remove();  
            queue.add(crntVal);  
        }  
    }  
  
    public int pop() {  
      return queue.remove();  
    }  
  
    public int top() {  
        return queue.peek();  
    }  
  
    public boolean empty() {  
        if(queue.size() > 0)  
            return false;  
  
        return true;  
    }  
}
