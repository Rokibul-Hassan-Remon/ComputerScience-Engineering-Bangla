public class MyStack{  
    Queue<Integer> queue;  
    public MyStack() {  
        queue = new ArrayDeque<>();  
    }  
  
    public void push(int x) {  
        Queue<Integer> temp = new ArrayDeque<>();  
        temp.add(x);  
  
        while(queue.size() > 0){  
            int crntVal = queue.remove();  
            temp.add(crntVal);  
        }  
       queue = temp;  
    }  
  
    public int pop() {  
      return queue.remove();  
    }  
  
    public int top() {  
        return queue.peek();  
    }  
  
    public boolean empty() {  
        if(queue.size() > 0)  
            return false;  
  
        return true;  
    }  
}
