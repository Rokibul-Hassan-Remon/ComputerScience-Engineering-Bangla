public static List<List<Integer>> combinationSum2(int[] candidates, int target) {  
    Arrays.sort(candidates);  
    List<Integer> path = new ArrayList<>();  
    combinationSum(candidates, 0, target, path);  
    
    return res;   
  
}   
  
public static List<List<Integer>> res = new ArrayList<>();  
public static void combinationSum(int[] candidates, int indx, int targetSum, List<Integer> path){  
    if(targetSum == 0){  
        res.add( new ArrayList<>(path) );  
          
        return;   
    }   
      
    for(int i = indx; i < candidates.length; i++){  
        if(i > indx && candidates[i] == candidates[i-1]) continue;  
        if(targetSum < candidates[i]) break;  
          
        path.add(candidates[indx]);  
        combinationSum(candidates, indx + 1, targetSum - candidates[i], path);  
        path.removeLast();   
    }   
}
