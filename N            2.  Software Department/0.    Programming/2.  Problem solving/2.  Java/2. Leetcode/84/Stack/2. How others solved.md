import java.util.Stack;  
  
class Solution {  
    public int MaximumRactangleArea(int[] array){  
        int arrayLength = array.length;  
        Stack<Integer> stack = new Stack<>();  
        int area = 0;  
  
        for(int i = 0; i < array.length; i++){  
  
            while(!stack.isEmpty() && array[stack.peek()] > array[i]){  
                int top = stack.pop();  
                if(!stack.isEmpty()){  
                    area = Math.max(area, array[i]*(i - stack.peek() -1));  
                }else {  
                    area = Math.max(area, array[top] * i);  
                }  
            }  
  
            stack.push(i);  
        }  
  
        while(!stack.isEmpty()){  
            int top = stack.pop();  
  
            if(!stack.isEmpty()){  
                area = Math.max(area, array[top]*(arrayLength - stack.peek() -1));  
            }else{  
                area = Math.max(area, array[top] * arrayLength);  
            }  
        }  
  
        return area;  
    }  
}
