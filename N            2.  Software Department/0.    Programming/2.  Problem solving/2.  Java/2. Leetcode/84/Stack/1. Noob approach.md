import java.util.Stack;  
  
class Solution {  
    public long MaximumRactangleArea(int[] array){  
    
        int arrayLength = array.length;  
        int[] leftMin = new int[arrayLength];  
        int[] rightMin = new int[arrayLength];  
        
        Stack<Integer> stack = new Stack<>();  
        int area = 0;  
        
        for(int i = 0; i < array.length; i++){  
            rightMin[i] = arrayLength;  
            
            while(!stack.isEmpty() && array[stack.peek()] > array[i]){  
                int pop = stack.pop();  
                rightMin[pop] = i ;  
            }  
            
            stack.push(i);  
        }  
        stack.clear();  
        
        for(int i = arrayLength - 1; i >= 0; i--){  
            leftMin[i] = -1;  
  
            while(!stack.isEmpty() && array[stack.peek()] > array[i]){  
                int pop = stack.pop();  
                leftMin[pop] = i ;  
            }  
  
            stack.push(i);  
        }  
  
        for(int i = 0; i < arrayLength; i++){  
            int width = (rightMin[i] - leftMin[i] -1);  
            int crntArea = width * array[i];  
            area = Math.max(crntArea, area);  
        }  
  
        return area;  
    }  
}
