import java.util.Stack;  
  
class StockSpanner {  
  
    private int indx = -1;  
    private Stack<int[]> stack;  
    public StockSpanner() {  
        stack = new Stack<>();  
    }  
  
    public int next(int price) {  
        indx++;  
        int result = 0;  
        while(!stack.isEmpty() && stack.peek()[0] <= price){  
           stack.pop();  
        }  
  
        if(!stack.isEmpty()){  
            int[] popArray = stack.peek();  
            result = indx - popArray[1];  
        }else{  
            result = indx+1;  
        }  
  
        stack.push(new int[]{price, indx});  
  
        return result == 0? 1 : result;  
    }  
}
