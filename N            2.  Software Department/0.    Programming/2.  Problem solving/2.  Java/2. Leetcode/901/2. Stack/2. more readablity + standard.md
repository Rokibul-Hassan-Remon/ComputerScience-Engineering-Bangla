import java.util.ArrayDeque;
import java.util.Deque;

class StockSpanner {
    private int day = -1;  
    private final Deque<int[]> stack; // each element = {price, index}
    
    public StockSpanner() {
        stack = new ArrayDeque<>();
    }
    
    public int next(int price) {
        day++;
        
        // Pop all previous days with price <= today's price
        while (!stack.isEmpty() && stack.peek()[0] <= price) {
            stack.pop();
        }
        
        // If stack empty → all previous days smaller
        int span = stack.isEmpty() ? day + 1 : day - stack.peek()[1];
        
        // Push current price and day
        stack.push(new int[]{price, day});
        
        return span;
    }
}
