class Solution {  
    public int[] nextGreaterElement(int[] nums1, int[] nums2) {  
        int[] result = new int[nums1.length ];  
  
        for(int i = 0; i < nums1.length; i++){  
            int nxtBigNumber = -1;  
            int flg = 0;  
  
            for(int j = 0; j < nums2.length; j++){  
                if(nums1[i] == nums2[j] && flg == 0){  
                    flg = 1;  
                }  
                if(nums1[i] < nums2[j] && flg == 1){  
                    nxtBigNumber = nums2[j];  
                    break;  
                }  
            }  
  
            result[i] = nxtBigNumber;  
        }  
  
        return result;  
    }  
}
