public  List<List<Integer>> permuteUnique(int[] nums) {
    List<List<Integer>> res = new ArrayList<>();
    Arrays.sort(nums);   // optional but recommended
    backtrack(nums, 0, res);
    
    return res;
}

public  void backtrack(int[] nums, int index, List<List<Integer>> res) {
    if (index == nums.length) {
        List<Integer> temp = new ArrayList<>();
        for (int n : nums) temp.add(n);
        res.add(temp);
        
        return;
    }

    HashSet<Integer> tracked = new HashSet<>();

    for (int i = index; i < nums.length; i++) {
        if (tracked.contains(nums[i])) continue;

        tracked.add(nums[i]);
        swap(nums, index, i);
        backtrack(nums, index + 1, res);
        swap(nums, index, i);
    }
}

public  void swap(int[] arr, int i, int j) {
    int t = arr[i];
    arr[i] = arr[j];
    arr[j] = t;
}
