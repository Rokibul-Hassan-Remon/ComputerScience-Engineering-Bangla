	public List<List<Integer>> combine(int n, int k) {  
	    int crntNumber = 1;  
	    List<List<Integer>> res = new ArrayList<>();  
	    List<Integer> path = new ArrayList<>();  
	  
	    backtrack(n, k, crntNumber, k, path, res);  
	  
	    return res;  
	}  
	  
	public void backtrack(int n, int k, int crntNumber, int rstk, List<Integer> path, List<List<Integer>> res){  
	    if(rstk == 0){  
	        res.add(new ArrayList<>(path));  
	  
	        return;  
	    }  
	  
	    for(int i = crntNumber; i <= n; i++){  
	        path.add(i);  
	        rstk--;  
	  
	        backtrack(n, k, i+1, rstk, path, res);  
	  
	        path.removeLast();  
	        rstk++;  
	    }  
	}
