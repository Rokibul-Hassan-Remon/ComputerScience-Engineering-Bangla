public class MyQueue {  
    Stack<Integer> inStack;  
    Stack<Integer> outStack;  
  
    public MyQueue() {  
        inStack = new Stack<>();  
        outStack = new Stack<>();  
    }  
  
    public void push(int x) {  
        inStack.push(x);  
    }  
  
    public int pop() {  
        shift();  
        return outStack.pop();  
    }  
  
    public int peek() {  
        shift();  
        return outStack.peek();  
    }  
  
    public boolean empty() {  
        shift();  
        return outStack.isEmpty();  
    }  
  
    public void shift() {  
        if (outStack.isEmpty()) {  
            while (!inStack.isEmpty()) {  
                outStack.push(inStack.pop());  
            }  
        }  
    }  
}
