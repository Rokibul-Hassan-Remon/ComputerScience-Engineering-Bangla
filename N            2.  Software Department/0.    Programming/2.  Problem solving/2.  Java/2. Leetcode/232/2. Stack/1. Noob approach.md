class MyQueue {  
    Stack<Integer> stack;  
    public MyQueue() {  
        stack = new Stack<>();  
    }  
  
    public void push(int x) {  
        stack.push(x);  
    }  
  
    public int pop() {  
        Stack<Integer> temp = new Stack<>();  
        while(stack.size() > 1){  
            int crnt = stack.pop();  
            temp.push(crnt);  
        }  
  
        int x = stack.pop();  
        //temp.push(x);  
        while(!temp.isEmpty()){  
            int crnt = temp.pop();  
            stack.push(crnt);  
        }  
  
        return x;  
    }  
  
    public int peek() {  
        Stack<Integer> temp= new Stack<>();  
        while(stack.size() > 1){  
            int crnt = stack.pop();  
            temp.push(crnt);  
        }  
  
        int x = stack.pop();  
        temp.push(x);  
        while(!temp.isEmpty()){  
            int crnt = temp.pop();  
            stack.push(crnt);  
        }  
  
        return x;  
    }  
  
    public boolean empty() {  
        return  stack.isEmpty();  
    }  
}
