public List<List<Integer>> subsets(int[] nums) {  
    Arrays.sort(nums);  
  
    int indx = 0;  
    List<Integer> path = new ArrayList<>();  
    List<List<Integer>> res = new ArrayList<>();  
  
    backtrck(nums, indx, path, res);  
  
    return res;  
}  
  
public void backtrck(int[] nums, int indx, List<Integer> path, List<List<Integer>> res){  
  
    res.add(new ArrayList<>(path));  
  
    for(int i = indx; i < nums.length; i++){  
  
        if( i > indx && nums[i] == nums[i-1] )  
            continue;  
  
        path.add(nums[i]);  
        backtrck(nums, i+1, path, res);  
        path.removeLast();  
    }  
}
